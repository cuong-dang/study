\subsection{Total Probability Theorem and Bayes' Rule}
\begin{theorem*}[Total Probability Theorem]
	Let $A_1,...,A_n$ be disjoint events that form a partition of the sample space (each possible outcome
	is included in exactly one of the events $A_1,...,A_n$) and assume that $P(A_i)>0$, for all $i$. Then,
	for any event $B$, we have
	\begin{align*}
		P(B) & = P(A_1\cap B)+\cdots+P(A_n\cap B)      \\
		P(B) & = P(A_1)P(B|A_1)+\cdots+P(A_n)P(B|A_n).
	\end{align*}
\end{theorem*}
